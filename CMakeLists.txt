cmake_minimum_required(VERSION 3.24)
project(RE4OGAimWalk LANGUAGES C CXX)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "Configure an x86 build; bio4.exe is 32-bit.")
endif()
add_subdirectory(third_party/minhook)
add_library(RE4OGAimWalk MODULE src/aim_walk.cpp)
target_compile_features(RE4OGAimWalk PRIVATE cxx_std_17)
target_compile_definitions(RE4OGAimWalk PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
if(MSVC)
  target_compile_options(RE4OGAimWalk PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(RE4OGAimWalk PRIVATE -Wall -Wextra -Wpedantic)
  target_link_options(RE4OGAimWalk PRIVATE -static -static-libgcc -static-libstdc++)
endif()
target_link_libraries(RE4OGAimWalk PRIVATE minhook user32 comctl32)
set_target_properties(RE4OGAimWalk PROPERTIES PREFIX "" SUFFIX ".asi" OUTPUT_NAME "RE4OGAimWalk")
